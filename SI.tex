\section{Descriptor definitions}
\label{si:defs}

\paragraph{Notation.} The cell of volume $V$ is divided into $N_v$ voxels $k$ of
volume $\Delta V=V/N_v$; $\rho_k$ is the density in e\,\AA$^{-3}$ (the
\texttt{CHGCAR} value divided by $V$), $Q_\mathrm{tot}=\sum_k\rho_k\Delta V$ and
$\langle\rho\rangle=Q_\mathrm{tot}/V$. For each voxel, $i(k)$ is the nearest
nucleus (periodic images included), $r_k$ its distance and $\hat{\mathbf u}_k$
the unit vector from that nucleus to the voxel. The core, bond and interstitial
shells are $r_k\le c_1$, $c_1<r_k\le c_2$ and $r_k>c_2$ (defaults $c_1=0.8$\,\AA,
$c_2=1.5$\,\AA). $\nabla\rho$, $\nabla^2\rho$ and the Hessian eigenvalues
$\lambda_1\le\lambda_2\le\lambda_3$ are computed spectrally on the periodic grid.
Quantities defined in atomic units (NCI, information measures) are evaluated in
bohr. A descriptor that is undefined for a special input returns a documented
constant and sets a flag.

\subsection{Bonding (19)}
\begin{description}
\item[\dn{m1}, \dn{m2}, \dn{sigma_r2}] [\AA, \AA$^2$, \AA$^2$]
$m_n=\sum_k\rho_k r_k^n/\sum_k\rho_k$, $\sigma_r^2=m_2-m_1^2$: mean distance of
the charge from its nearest nucleus, second moment and radial spread.
\item[\dn{f_core}, \dn{f_bond}, \dn{f_int}] fraction of the charge in the core,
bond and interstitial shells; they sum to one.
\item[\dn{zeta}] [0--1] $\zeta=1-\sum_k|\nabla\rho_k\cdot\hat{\mathbf u}_k|/\sum_k|\nabla\rho_k|$:
zero for spherical, non-overlapping atoms.
\item[\dn{lnf}] $N_v^{-1}\sum_k\mathbf 1[\nabla^2\rho_k<0]$: volume fraction where
charge is concentrated.
\item[\dn{lnf_charge_weighted}] $\sum_{\nabla^2\rho_k<0}\rho_k/\sum_k\rho_k$: the
same as a charge fraction.
\item[\dn{lap_concentration}] $\sum_{\nabla^2\rho_k<0}|\nabla^2\rho_k|/\sum_k|\nabla^2\rho_k|$;
identically $\tfrac12$ on a periodic grid, kept for completeness.
\item[\dn{lap_concentration_valence}] the same ratio over the voxels with
$r_k>c_1$.
\item[\dn{ellip_bond_bounded_avg}] [0--1) mean of $1-\lambda_2/\lambda_1=\varepsilon/(1+\varepsilon)$
over the bond-shell voxels with $\lambda_2<0$: zero for cylindrical charge.
\item[\dn{f_NCI}] volume fraction of the NCI region
$\{s_k<0.5,\ \rho_k<0.05$ a.u.$\}$, with
$s=|\nabla\rho|/\big(2(3\pi^2)^{1/3}\rho^{4/3}\big)$.
\item[\dn{NCI_attractive}] fraction of the NCI voxels with $\lambda_2<0$.
\item[\dn{sign_lambda2_rho_mean}] [e\,bohr$^{-3}$] mean of
$\mathrm{sign}(\lambda_2)\rho$ over the NCI region.
\item[\dn{V_spread}, \dn{V_int_min}] [eV] standard deviation of the Hartree
potential energy $V(\mathbf G)=4\pi k_e\rho(\mathbf G)/|\mathbf G|^2$ (cell
average zero) over the atomic sites; its minimum in the interstitial shell.
\item[\dn{rho_mid_mean}, \dn{rho_mid_std}] [e\,\AA$^{-3}$] mean and standard
deviation of $\rho$, by Fourier interpolation, at the midpoints of the
first-shell bonds ($|\mathbf R_{ij}|\le1.1\,d_i$, $d_i$ the nearest-neighbour
distance of atom $i$).
\end{description}

\subsection{Structural (19)}
\begin{description}
\item[\dn{rho_perc_a}, \dn{rho_perc_b}, \dn{rho_perc_c}] [e\,\AA$^{-3}$]
$\max\{c:\{\rho>c\}$ contains a cluster that winds around the cell along
$\mathbf a_\alpha\}$: density at the bottleneck of the best connecting path.
\item[\dn{perc_anisotropy}] $(\max_\alpha-\min_\alpha)\rho_{\mathrm{perc},\alpha}/\mathrm{mean}_\alpha\rho_{\mathrm{perc},\alpha}$.
\item[\dn{n_max}, \dn{n_min}] [\AA$^{-3}$] voxels higher (lower) than all 26
neighbours, per volume.
\item[\dn{n_NNM}, \dn{Q_NNM}] maxima farther than 0.8\,\AA{} from every nucleus,
per volume [\AA$^{-3}$]; charge in their steepest-ascent basins over
$Q_\mathrm{tot}$.
\item[\dn{rho_min}, \dn{rho_min_ratio}, \dn{rho_int_mean}] $\min_k\rho_k$
[e\,\AA$^{-3}$]; $\rho_{\min}/\langle\rho\rangle$; mean $\rho$ over the
interstitial shell [e\,\AA$^{-3}$].
\item[\dn{T_eigenvalues_t1}, \dn{_t2}, \dn{_t3}] sorted eigenvalues of
$T_{ab}=\sum_k\partial_a\rho_k\partial_b\rho_k/\sum_k|\nabla\rho_k|^2$ (trace 1;
all $\tfrac13$ for a cubic crystal).
\item[\dn{charge_FA}] [0--1] $\sqrt{3/2}\,\lVert T-\tfrac13I\rVert_F/\lVert T\rVert_F$.
\item[\dn{shannon_entropy}] $S=-\int\tilde\rho\ln\tilde\rho\,dV-\ln V$ with
$\tilde\rho=\rho/N_e$; zero for a uniform density.
\item[\dn{disequilibrium}] $D=V\int\tilde\rho^{\,2}\,dV$; one for a uniform density.
\item[\dn{LMC_complexity}] $C=De^{S}$.
\item[\dn{fisher_information}] [bohr$^{-2}$] $I=\int|\nabla\tilde\rho|^2/\tilde\rho\,dV$.
\end{description}

\subsection{Site heterogeneity (18)}
Each voxel is assigned to an atom with weight $w_i(k)$ (nearest atom by default;
power diagram or Becke cells optionally), and three per-atom quantities are
computed:
\begin{equation}
m_1^{(i)}=\frac{\sum_k w_i(k)\rho_k r_{ik}}{\sum_k w_i(k)\rho_k},\quad
f_\mathrm{bond}^{(i)}=\frac{\sum_{c_1<r_{ik}\le c_2}w_i(k)\rho_k}{\sum_k w_i(k)\rho_k},\quad
\zeta^{(i)}=1-\frac{\sum_k w_i(k)|\nabla\rho_k\cdot\hat{\mathbf u}_{ik}|}{\sum_k w_i(k)|\nabla\rho_k|}.
\end{equation}
For each $X\in\{m_1,f_\mathrm{bond},\zeta\}$ the descriptors are
\dn{X_site_std} ($\mathrm{std}_iX^{(i)}$), \dn{X_site_range}, \dn{X_site_max},
\dn{X_site_min}, \dn{X_within_element_var}
($\sum_e x_e\mathrm{Var}_{i\in e}X^{(i)}$) and \dn{X_between_element_var}
($\sum_e x_e(\bar X_e-\bar X)^2$), with $x_e$ the site fraction of element $e$;
the last two add up to $\mathrm{Var}_iX^{(i)}$. The within-element variance is NaN
(flagged) when no element has two or more sites, and the between-element
variance is zero (flagged) for an elemental crystal.

\section{Property models}
\label{si:props}

\paragraph{Targets.} The bulk, shear and Young's moduli (Voigt--Reuss--Hill),
the energy relative to the free atoms ($E_\mathrm{form}$ of the benchmark) and
its Debye temperature are taken from the property table of the cross-model
benchmark of S.~K.~Maurya and S.~Bhowmick (manuscript in preparation). All
other targets come from the Materials Project, database version 2026.04.13, for
the same material identifiers. Elastic data were kept only where $K$ and $G$ lie
between 0 and 1000\,GPa and agree with the benchmark table (6,000 materials);
$C_{11}$, $C_{12}$ and $C_{44}$ only where the smallest eigenvalue of the
unrounded stiffness tensor exceeds 1\,GPa (5,791). The direct-gap label is
defined only for materials with a gap (1,836).

\paragraph{Training.} Inputs: the 56 descriptors of pydemi without the constant
Laplacian concentration ratio (55). Ridge and logistic regression: median
imputation, standardization, penalty from $10^{-3}$ to $10^{3}$ (13 values)
chosen on the validation set. Random forests: 500 trees, $\frac13$ of the
features per split for regression and $\sqrt{n}$ for classification, median
imputation. LightGBM: missing values handled natively; 50 Optuna TPE trials
(sampler seed 0) over learning rate 0.01--0.1 (log), 7--127 leaves (log),
minimum leaf size 5--100 (log), feature fraction 0.3--1, bagging fraction
0.5--1 and $L_1$, $L_2$ penalties $10^{-8}$--10 (log); objective the squared
error of the (transformed) target or the log-loss; early stopping after 200
rounds without improvement on the validation set, at most 10,000 rounds.
$\log_{10}$ was applied to the refractive index. Three of the 38 LightGBM configurations (with the 55 descriptors, the cohesive energy, the Debye temperature and the Cahill conductivity) used more than 9,500 rounds on average, so a larger cap could improve them slightly. SHAP values are the
exact tree contributions of LightGBM on the test set, averaged over the seeds.

\paragraph{Ten descriptors per property.} For each property, the 55 descriptors
were ranked by their mean absolute SHAP value in the five LightGBM models on the
\emph{training} set, averaged over the seeds; the ten highest
(Table~\ref{tab:si_top10}) were the only inputs of a second set of ridge, random
forest and LightGBM models, trained and tuned exactly as above
(Table~\ref{tab:si_full}).

\paragraph{Correlated descriptors.} On the 4,847 training structures, the
Spearman correlation $\rho_s$ was computed between all pairs of the 55
descriptors. Descriptors were clustered by complete linkage on $1-|\rho_s|$ with
a threshold of 0.10, so that every pair in a cluster has $|\rho_s|\geq0.90$;
in each of the 11 clusters with more than one member, the descriptor with the
largest mean SHAP share over the 17 properties (training set) was kept and the
other 18 were dropped (Table~\ref{tab:si_redundant_groups}). The 17 properties
were retrained on the remaining 37 descriptors exactly as above. The change in
test score and its 95\% interval come from a paired bootstrap of the test
materials (2,000 resamples) on the predictions averaged over the five seeds
(Table~\ref{tab:si_redundant_scores}).

\paragraph{Reference densities.} Most library tests use densities whose exact
values are known: superpositions of normalized Gaussians,
$\rho(r)=N(\alpha/\pi)^{3/2}e^{-\alpha r^2}$, and of 1s Slater functions,
$\rho(r)=N\zeta^3\pi^{-1}e^{-2\zeta r}$, summed over periodic images and
evaluated on the grid together with their analytic gradient and Hessian
(\texttt{pydemi.validate.analytic}). The Gaussians are smooth and band-limited
on fine grids; the Slater functions have the nuclear cusp of a real density.
Most test cells are triclinic, so that terms that vanish for orthogonal axes are
exercised.

\paragraph{Continuous integration.} Every push to the repository runs 14 jobs
on GitHub Actions: the full suite with the GUI and property extras on Linux,
macOS and Windows with Python 3.10 and 3.13, and on Linux with 3.11 and 3.12 (8
jobs); the suite without any extra on Linux, which checks that the library
needs only NumPy, SciPy and pandas and that \texttt{pydemi gui} and
\texttt{pydemi predict} then say which extra to install; the density-model,
interface and property tests with the ML extra (CPU PyTorch) on all three
systems (3 jobs); a static type check (\texttt{mypy --strict}); and a build of
the source distribution and wheel, which installs the wheel in a clean
environment and starts the interface.

\paragraph{Skipped tests.} Tests skip, with the reason, when an optional
dependency is missing: the interface tests without Streamlit and Plotly, the
network tests without PyTorch and e3nn, the structure-file tests without ASE,
the property-model tests without LightGBM, and one input test without pymatgen.
One test reads a CHGCAR of the dataset (FeNi$_3$) to check that the NCI fields
stay finite on a real, partly negative PAW pseudo-density; it runs where the
dataset is available and skips elsewhere. The three tests that send signals to
the interface server are skipped on Windows.

\paragraph{Invariance harness (169).} A Fe$_2$O crystal in a triclinic cell
(three Gaussians on a $20\times21\times22$ grid) is featurized as it is, as its
$2\times2\times2$ supercell, with its origin translated by $(3,7,11)$ voxels and
rigidly rotated by 37$^\circ$ about $(1,2,0.5)$. Each of the 56 descriptors must
agree under each of the three transformations to
$|a-b|\leq10^{-6}\max(|a|,|b|)+10^{-12}$ (168 tests). The tests are generated
from the descriptor registry, so a descriptor added later is checked without a
new test. One more test confirms that the transformations do what they claim:
the supercell has 8 times the atoms, volume and charge, and the rotation
preserves the volume and the metric tensor.

\paragraph{Derivatives (16).} Spectral (FFT) gradient, Laplacian and Hessian of
a Gaussian superposition are exact to $10^{-10}$ relative in an orthorhombic and
a triclinic cell. Finite differences of order 2, 4 and 6 reach their nominal
convergence order between $32^3$ and $64^3$ grids (observed order at least the
nominal order minus 0.6) in both cells. The Laplacian built from the diagonal of
the metric is exact for the orthorhombic cell but more than 5\% wrong for the
triclinic one, which is why the full metric is used. The Hessian is symmetric,
its trace equals the Laplacian to $10^{-9}$, its eigenvalues are ascending and
match the analytic ones ($10^{-9}$ with FFT, $5\times10^{-3}$ with finite
differences), and computing them in chunks changes nothing. For the cusped
Slater density, finite differences converge away from the nuclei ($r>0.5$\,\AA:
errors below $10^{-3}$ for the gradient and $5\times10^{-3}$ for the Laplacian
on $96^3$, decreasing more than eightfold from $48^3$), while the FFT Laplacian
rings across the cell and does not converge; this documented limitation is
tested so that it cannot change unnoticed. One test checks $|\mathbf G|^2$
against the spectral Laplacian and one that invalid options are rejected.

\paragraph{Geometry and partitions (18).} Nearest-atom distances, owners and
directions agree with a brute-force search over $\pm5$ periodic images to
$10^{-12}$\,\AA{}, also in an extremely sheared cell in which a fixed
$3\times3\times3$ supercell misses the nearest image by more than 0.01\,\AA{};
the power-diagram assignment agrees with brute force in the same way. A voxel on
a nucleus gets a zero direction; the core, bond and interstitial shells cover
every voxel exactly once, and shells scaled by the covalent radii use them; and the
geometry of a structure is computed once and shared (6 tests). The nearest-atom,
power and Becke partitions have weights that sum to one at every voxel and
conserve the charge to $10^{-12}$; two symmetry-equivalent atoms receive equal
Becke charges to $10^{-5}$ (the products are truncated to the nearest images); the
two-atom Becke weight satisfies $s(\mu)+s(-\mu)=1$ with $s(0)=\tfrac12$; the site
descriptors depend on the chosen partition and record it; and the site
descriptors from the power and Becke partitions are invariant under the three
transformations of the harness to $10^{-6}$ (12 tests).

\paragraph{Closed forms and identities (40).} \emph{Bonding} (12): for a single
Slater density ($\zeta=3$\,\AA$^{-1}$, $96^3$ grid, 4\,\AA{} box) the charge
integrates to 1 ($10^{-4}$), $m_1=3/(2\zeta)$ ($5\times10^{-4}$), $m_2=3/\zeta^2$
($1.5\times10^{-3}$, limited by nearest-image distances in the small box),
$\sigma_{r^2}$ within 1\% and the anisotropy $\zeta_\mathrm{anis}<5\times10^{-5}$,
with both derivative methods; the error of $m_1$ decreases on refining the grid;
the negative-Laplacian volume fraction equals that of the sphere $r<1/\zeta$
within 5\% with finite differences, whereas the ringing FFT Laplacian
overestimates it by more than 10\%; two atoms give a non-zero anisotropy; the
Laplacian concentration is $\tfrac12$ to $10^{-10}$ for any density because
$\int\nabla^2\rho\,\mathrm dV=0$ on a periodic grid; a uniform density gives the
documented constants with their flags and no NaN; names, order, metadata and the
catalogue are generated from the registry, which refuses extensive quantities;
and invalid options are rejected. \emph{Structural} (11): the function
$\cos2\pi x+\cos2\pi y+\cos2\pi z$ with a small periodic tilt has exactly one
maximum, one minimum and $3+3$ saddles per cell, with Euler characteristic
consistency 0, on $12^3$, $24^3$ and $48^3$ grids, and the counts are reported
per volume; a simple-cubic lattice of Gaussians percolates exactly at the
bond-midpoint density in all three directions ($10^{-12}$, anisotropy 0); the
percolation threshold does not depend on the cell origin, needs a cluster that
winds around the cell, and is the highest spanning level; a non-nuclear maximum
placed between two atoms is counted with its basin charge; a uniform density
gives zero Shannon entropy, unit disequilibrium and LMC complexity and an
isotropic anisotropy tensor (flagged); and a layered density gives
$t_3=1$ and fractional anisotropy 1. \emph{Fields} (6): the Hartree potential of a
Gaussian in a cubic box matches $kN[\mathrm{erf}(\sqrt\alpha r)/r-2\sqrt{\alpha/\pi}]$
plus the $r^2$ term of the neutralizing background to $2\times10^{-3}$; the
potential is referenced to its cell average ($10^{-12}$); the NCI region between
two distant atoms has fractions between 0 and 1, and a uniform density gives the
NCI sentinel; the bond-midpoint density of two Gaussians matches its closed form
to $10^{-6}$; and the NCI fields are finite on a real CHGCAR (above).
\emph{Site statistics} (6): the within- and between-element variances add up to
the total variance to $10^{-12}$, for random numbers and for computed site
values; the site charges add up to the cell charge and the site moments to
$m_1$ ($10^{-12}$); range, extrema and standard deviation are consistent; the
within-element variance is NaN (flagged) when no element has two sites and the
between-element variance is zero (flagged) for one element; and the power
partition gives a different tiling with the same total charge. \emph{Bond
ellipticity} (5): the bounded ellipticity $1-\lambda_2/\lambda_1$ is in the
default output, equals the value from the exact Hessian of an anisotropic
Gaussian (two shapes) to $10^{-3}$, vanishes for a spherical one ($10^{-6}$) and is the
monotone map $\epsilon/(1+\epsilon)<1$ of the usual ellipticity.

\paragraph{Input, batch and tooling (21).} \emph{Input} (10): the lattice
conventions ($\mathbf a_i\cdot\mathbf b_j=\delta_{ij}$, metric, volume; singular
cells rejected); a density on a different lattice is rejected; a pymatgen
structure is accepted; a written CHGCAR reads back to $10^{-10}$; the stored
$\rho V$ is divided by the cell volume; the Fortran order of the values
(fastest index first) is preserved; spin-polarized (2 blocks) and non-collinear
(4 blocks) files with augmentation occupancies and magnetic moments give the
total density from the first block; a VASP~4 file with a negative scale factor
(the volume), Cartesian coordinates and selective dynamics reads correctly once
the species are given; and POTCAR-style labels such as \texttt{Fe\_pv} are
reduced to the element. \emph{Tooling} (9): a batch with a broken file keeps a
fixed column schema, records the error and continues, with one and with two
worker processes, or skips or stops on request; the four command-line commands
(\texttt{featurize}, \texttt{catalogue}, \texttt{batch}, \texttt{sweep}) write
the expected files; the shell-radius sweep skips invalid pairs and its shell
fractions add up to 1; the grid-convergence report works (here $m_1$ changes by
less than $10^{-3}$ on an 80\% grid); the recommended grid spacing, 0.08\,\AA{}
for 2\%, is derived from the Slater closed forms; single precision agrees with
double precision (median relative difference below $10^{-5}$, at most 5\% of
the descriptors above $10^{-3}$); and a uniform density gives no bare NaN in any
domain. \emph{Regressions} (2): an object reused with other options (metric or
diagonal Laplacian, finite-difference order 2 or 6) gives the same values as a
fresh one ($10^{-12}$), because derived fields are cached per option set.

\paragraph{Density model (13).} Without PyTorch: the VASP grid rule (the
$40^3$ grid of the BeCo CHGCAR of the dataset; even sizes with prime factors 2,
3, 5 and 7; fewer points at a lower cut-off); the order of the grid points; the
POTCAR table (valence electrons of the 87 elements of the training set; other
elements reported); atom--atom and probe--atom neighbour lists in a triclinic
cell equal to a brute-force search over $\pm3$ images; and the chunk size that
follows the memory budget (5 tests). With PyTorch and e3nn: the checkpoint loads
(SHA-256 and 1,909,389 parameters); its prediction for BeCo on an $8^3$ grid
equals the one stored from the original ChargE3Net code (PyTorch 1.13, e3nn 0.5)
to $10^{-5}$ of the maximum density; predictions do not depend on the chunk size
and are invariant to rotating the cell, permuting the atoms and shifting them by
lattice vectors ($10^{-4}$ relative); the predicted density is rescaled to the
valence electron count ($10^{-10}$) and records its provenance (raw electron
count, scale factor, grid); a prediction can be cancelled; and elements outside
the training set are refused (6 tests). With ASE: CIF and POSCAR files are read,
and \texttt{pydemi predict} writes a CHGCAR with the right electron count (2 tests).

\paragraph{Property models (12).} A small bundle in the distributed format
(LightGBM models on 200 random descriptor vectors, among them a regression on
$\log_{10}$ and a classification) is built for the tests. Predictions reproduce
the stored test predictions to $10^{-9}$, including the logarithmic transform;
predictions are finite, clipped to physical ranges (dielectric constant
$\geq1$, probabilities in $[0,1]$) and allow NaN descriptors; dataset materials are
recognized from their descriptors and out-of-range inputs flagged; the model
path, a missing model and a LightGBM that cannot load (macOS without OpenMP) are
reported clearly; and the tables, figures and texts of the property view are
built. The distributed bundle of 22 models reproduces the test-set predictions
made with the untrimmed models when it was built, from the stored descriptors
and formulas, to $10^{-9}$ (every 30th test material).

\paragraph{Graphical interface (63).} \emph{Input} (10): a CHGCAR reads to the
same density plain or compressed (gzip, bzip2, xz, zip); unreadable files and
unknown partitions give clear errors; the summary (formula, lattice, electron
count) and the result table cover all 56 descriptors; and every descriptor has
a typeset formula. \emph{Descriptor figures} (10, each with the nearest-atom and
the Becke partition): every descriptor has a figure, and the plotted data
reproduce the descriptors --- the radial charge distribution gives the shell
fractions, the negative-Laplacian fractions, the Laplacian concentration and the
Fisher information to $10^{-9}$, the histograms are centred on $\zeta$, the
anisotropy-tensor eigenvalues and the bond ellipticity within one bin, and the
critical points, NNM charge, site potentials, bond midpoints, percolation
levels and site statistics give their descriptors to $10^{-7}$--$10^{-12}$; the
figure data stay below 2\,MB. \emph{Viewer} (11): large grids are subsampled to
actual grid values; slices pick the nearest lattice plane periodically; slices
along each axis on linear and logarithmic scales serialize; and atoms on the
plane are marked at all periodic images. \emph{Launcher} (7): \texttt{pydemi gui}
serves on the local machine only, ships the descriptor reference, explains how
to install a missing extra, opens the first URL, and never leaves the server
running after the launcher receives SIGTERM, SIGHUP or SIGINT, even when the
server ignores them. \emph{Background predictions} (4): one prediction runs at a
time with progress, can be cancelled and restarted, reports errors, keeps a few
finished results and can use a given grid. \emph{Compare view} (5): the relative
difference, the per-descriptor transfer table, the marking of descriptors that
differ, are near zero in both densities or are constant, the density error
(NMAPE), which requires identical grids, and the agreement of properties within
the typical error of each model. \emph{End to end} (16): the interface is run
headless (Streamlit AppTest) from upload to export --- the empty state, the
summary after an upload, the structure view, the descriptor table with search,
export and partition choice, the figure browser, views that keep their settings,
the property view with and without a model or LightGBM, unreadable uploads, the
structure tab without the ML extra or with an unsupported element, a ChargE3Net
prediction from a CIF followed by its descriptors, and the Compare view, which
predicts the density on the CHGCAR's own grid and shows both sets of
descriptors and properties side by side.
